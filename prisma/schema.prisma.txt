generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Aluno {
  id        String     @id @default(uuid())
  nome      String
  turma     String
  fone      String
  qrcode    String     @unique
  cafe      Boolean    @default(false)
  almoco    Boolean    @default(false)
  criadoEm  DateTime   @default(now())
  historicos Historico[]
}

model Historico {
  id       String   @id @default(uuid())
  alunoId  String?
  aluno    Aluno?   @relation(fields: [alunoId], references: [id], onDelete: SetNull)
  nome     String
  turma    String
  fone     String
  qrcode   String
  data     String
  hora     String
  cafe     Boolean  @default(false)
  almoco   Boolean  @default(false)
  criadoEm DateTime @default(now())
}

model Config {
  chave String @id
  valor String @db.Text
}
